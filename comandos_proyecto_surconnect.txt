    network:
      version: 2
      ethernets:
        ens33:

ip addr show

# Probar conectividad
ping -c 4 192.168.137.1
ping -c 4 192.168.137.2
ping -c 4 8.8.8.8

# Probar DNS
nslookup surconnect.local

# Actualizar sistema
sudo apt update
sudo apt upgrade -y

# Instalar Nginx
sudo apt install -y nginx

# Verificar que Nginx se instaló
nginx -v

# Ver estado de Nginx
sudo systemctl status nginx

# Generar certificado SSL autofirmado para HTTPS
sudo openssl req -x509 -nodes -days 365 -newkey rsa:2048 \
  -keyout /etc/ssl/private/surconnect.key \
  -out /etc/ssl/certs/surconnect.crt \
  -subj "/CN=iniciales.net/O=SurConnect IT Solutions/C=CO"

# Verificar que se crearon los archivos
ls -la /etc/ssl/private/surconnect.key
ls -la /etc/ssl/certs/surconnect.crt

# Crear archivo de configuración del sitio
sudo nano /etc/nginx/sites-available/surconnect

    Contenido:
    server {
        listen 443 ssl;
        server_name iniciales.net;

        ssl_certificate /etc/ssl/certs/surconnect.crt;
        ssl_certificate_key /etc/ssl/private/surconnect.key;

        root /var/www/surconnect;
        index index.html;

        location / {
            try_files $uri $uri/ =404;
        }
    }

    server {
        listen 80;
        server_name iniciales.net;
        return 301 https://$server_name$request_uri;
    }

# Crear directorio para el sitio web
sudo mkdir -p /var/www/surconnect

# Crear página web del portal
sudo nano /var/www/surconnect/index.html

    Contenido:
    <!DOCTYPE html>
    <html>
    <head>

            h2 { font-size: 32px; margin-bottom: 15px; }
            p { font-size: 20px; line-height: 1.6; }
        </style>
    </head>
    <body>
        <h1>SurConnect IT Solutions</h1>
        <h2>Portal Corporativo</h2>
        <p>Bienvenido al portal de SurConnect</p>
        <p>Servicios de outsourcing de TI para Sudamérica</p>
        <p>Desarrollo de software | Ciberseguridad | Cloud Computing</p>
    </body>
    </html>

# Dar permisos correctos al directorio web
sudo chown -R www-data:www-data /var/www/surconnect
sudo chmod -R 755 /var/www/surconnect
